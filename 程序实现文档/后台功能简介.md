export const getUsers = async (req: AuthRequest, res: Response): Promise<void> => {
  const page = parseInt(req.query.page as string) || 1;
  const limit = parseInt(req.query.limit as string) || 10;
  const username = req.query.username as string;
  const nickname = req.query.nickname as string;
  const role = req.query.role as string;
  const is_active = req.query.is_active as string;

  // 构建动态WHERE子句
  let whereClause = 'WHERE 1=1';
  const queryParams: any[] = [];

  if (username) {
    whereClause += ' AND u.username LIKE ?';
    queryParams.push(`%${username}%`);
  }
  if (nickname) {
    whereClause += ' AND u.nickname LIKE ?';
    queryParams.push(`%${nickname}%`);
  }
  if (role) {
    whereClause += ' AND r.code = ?';
    queryParams.push(role);
  }
  if (is_active !== undefined && is_active !== '') {
    whereClause += ' AND u.is_active = ?';
    queryParams.push(parseInt(is_active));
  }

  const [rows] = await db.execute(`
    SELECT u.*, r.name as role_name, r.code as role
    FROM users u
    LEFT JOIN roles r ON u.role_id = r.id
    ${whereClause}
    LIMIT ${limit} OFFSET ${(page - 1) * limit}
  `, queryParams);

  res.json({ success: true, data: rows });
};
```

export const createUser = async (req: AuthRequest, res: Response): Promise<void> => {
  const { username, password, email, nickname, phone, role_id, gender, birth_date } = req.body;
  const currentUserRole = req.user?.role;

  // 权限检查：只有超级管理员可以创建超级管理员
  if (role_id === 3 && currentUserRole !== 'super_admin') {
    res.status(403).json({
      success: false,
      message: '只有超级管理员可以创建超级管理员账号'
    });
    return;
  }

  // 检查用户名是否已存在
  const [existingUsers] = await db.execute(
    'SELECT id FROM users WHERE username = ?',
    [username]
  );
  if ((existingUsers as any[]).length > 0) {
    res.status(400).json({ success: false, message: '用户名已存在' });
    return;
  }

  // 密码加密 (bcrypt 12轮)
  const hashedPassword = await bcrypt.hash(password, 12);

  // 插入用户
  const [result] = await db.execute(
    `INSERT INTO users (username, password_hash, email, nickname, phone, role_id, gender, birth_date, is_active)
     VALUES (?, ?, ?, ?, ?, ?, ?, ?, 1)`,
    [username, hashedPassword, email, nickname, phone, role_id, gender, birth_date]
  );

  res.status(201).json({
    success: true,
    message: '用户创建成功',
    data: { userId: (result as any).insertId }
  });
};
```

export const updateRoleMenus = async (req: AuthRequest, res: Response): Promise<void> => {
  const { roleId } = req.params;
  const { menuIds } = req.body;

  // 开启事务
  const connection = await db.getConnection();
  await connection.beginTransaction();

  try {
    // 删除旧的权限关联
    await connection.execute('DELETE FROM role_menus WHERE role_id = ?', [roleId]);

    // 插入新的权限关联
    if (menuIds && menuIds.length > 0) {
      const values = menuIds.map((menuId: number) => [roleId, menuId]);
      await connection.query(
        'INSERT INTO role_menus (role_id, menu_id) VALUES ?',
        [values]
      );
    }

    await connection.commit();
    res.json({ success: true, message: '角色权限更新成功' });
  } catch (error) {
    await connection.rollback();
    res.status(500).json({ success: false, message: '更新失败' });
  } finally {
    connection.release();
  }
};
```

function getLocalizedField(zhValue: string, enValue: string | null, lang: string): string {
  if (lang.startsWith("en")) {
    return enValue || zhValue; // 英文优先,无则回退中文

export const getFoods = async (req: AuthRequest, res: Response): Promise<void> => {
  const page = parseInt(req.query.page as string) || 1;
  const limit = parseInt(req.query.limit as string) || 10;
  const lang = req.headers['accept-language'] || 'zh-CN';

  let whereClause = 'WHERE is_active = true';
  const queryParams: any[] = [];

  if (req.query.search) {
    whereClause += ' AND (name LIKE ? OR name_en LIKE ?)';
    const searchPattern = `%${req.query.search}%`;
    queryParams.push(searchPattern, searchPattern);
  }

  if (req.query.category) {
    whereClause += ' AND category = ?';
    queryParams.push(req.query.category);
  }

  const [rows] = await db.execute(
    `SELECT * FROM foods ${whereClause} LIMIT ${limit} OFFSET ${(page - 1) * limit}`,
    queryParams
  );

  // 本地化字段处理
  const localizedFoods = (rows as any[]).map(food => ({
    ...food,
    name: getLocalizedField(food.name, food.name_en, lang),
    category: getLocalizedField(food.category, food.category_en, lang)
  }));

  const targetUserId = getTargetUserId(req, res); // 获取目标用户ID

      AVG(weight) as avg_weight,
      AVG(exercise_duration) as avg_exercise_duration,
      AVG(sleep_hours) as avg_sleep_hours
     FROM health_records
     WHERE user_id = ? AND record_date >= ?`,
    [targetUserId, startDateStr]

  // 饮食记录统计

    `SELECT
      COUNT(*) as count,

    endDateStr = req.query.endDate as string;
  } else {
    const days = parseInt(req.query.days as string) || 30;
    const startDate = new Date();
    startDate.setDate(startDate.getDate() - days);
    startDateStr = startDate.toISOString().split('T')[0];
    endDateStr = new Date().toISOString().split('T')[0];
  }

  const [rows] = await db.execute(
    `SELECT record_date, weight
     FROM health_records
     WHERE user_id = ? AND record_date >= ? AND record_date <= ? AND weight IS NOT NULL
     ORDER BY record_date ASC`,
    [targetUserId, startDateStr, endDateStr]
  );

  res.json({ success: true, data: rows });
};
```

#### 营养分析

```typescript
export const getNutritionAnalysis = async (req: AuthRequest, res: Response): Promise<void> => {
  const targetUserId = getTargetUserId(req, res);
  const days = parseInt(req.query.days as string) || 7;

  const [rows] = await db.execute(
    `SELECT
      SUM(calories) as total_calories,
      SUM(protein) as total_protein,
      SUM(fat) as total_fat,
      SUM(carbs) as total_carbs
     FROM diet_records
     WHERE user_id = ? AND record_date >= DATE_SUB(CURDATE(), INTERVAL ? DAY)`,
    [targetUserId, days]
  );

  const data = rows[0];
  const totalProtein = parseFloat(data.total_protein) || 0;
  const totalFat = parseFloat(data.total_fat) || 0;
  const totalCarbs = parseFloat(data.total_carbs) || 0;

  // 计算热量占比
  const proteinCalories = totalProtein * 4;
  const fatCalories = totalFat * 9;
  const carbsCalories = totalCarbs * 4;
  const totalCaloriesFromMacros = proteinCalories + fatCalories + carbsCalories;

  res.json({
    success: true,
    data: {
      period_days: days,
      total_calories: parseFloat(data.total_calories) || 0,
      total_protein: totalProtein,
      total_fat: totalFat,
      total_carbs: totalCarbs,
      protein_percentage: ((proteinCalories / totalCaloriesFromMacros) * 100).toFixed(1),
      fat_percentage: ((fatCalories / totalCaloriesFromMacros) * 100).toFixed(1),
      carbs_percentage: ((carbsCalories / totalCaloriesFromMacros) * 100).toFixed(1)
    }
  });
};
```

    }
