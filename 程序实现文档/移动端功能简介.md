```typescript
export function useLoginForm() {
  const router = useRouter()
  const userStore = useUserStore()
  const { t } = useI18n()

  const formData = ref({
    username: '',
    password: '',
    remember: false
  })

  async function handleLogin() {
    loading.value = true
    try {
      // 调用登录API
      const response = await postAuthLogin({
        username: formData.value.username,
        password: formData.value.password
      })

      const data = (response as any).data
      if (data.token) {
        // 保存token
        const storage = formData.value.remember ? localStorage : sessionStorage
        storage.setItem('token', data.token)

        // 设置用户状态
        userStore.setToken(data.token)
        userStore.setUserInfo(data.user)

        showSuccessToast(t('login.loginSuccess'))

        // 跳转逻辑:首次登录跳转目标页,否则跳转首页
        const hasSetGoals = localStorage.getItem('hasSetGoals')
        if (!hasSetGoals) {
          router.replace('/goals?firstTime=true')
        } else {
          router.replace('/home')
        }
      }
    } catch (error: any) {
      showToast(error.message || t('login.loginFailed'))
    } finally {
      loading.value = false
    }
  }

  return { formData, loading, handleLogin }
}
```

### 用户状态管理

**Store**: `stores/user.ts:1-109`

核心功能:
- 用户信息和token管理
- localStorage/sessionStorage双存储
- 自动刷新用户信息

```typescript
export const useUserStore = defineStore('user', () => {
  const userInfo = ref<API.User | null>(null)
  const token = ref<string | null>(null)

  // 计算属性
  const isLogin = computed(() => !!token.value && !!userInfo.value)
  const username = computed(() => userInfo.value?.username || '用户')
  const nickname = computed(() => userInfo.value?.nickname || userInfo.value?.username || '用户')

  /**
   * 设置用户信息并同步到 localStorage
   */
  function setUserInfo(info: API.User) {
    userInfo.value = info
    localStorage.setItem('userInfo', JSON.stringify(info))
  }

  /**
   * 设置 token
   */
  function setToken(newToken: string) {
    token.value = newToken
  }

  /**
   * 初始化用户信息（从 localStorage 恢复）
   */
  function initUserInfo() {
    // 恢复 token (支持localStorage和sessionStorage)
    const savedToken = localStorage.getItem('token') || sessionStorage.getItem('token')
    if (savedToken) {
      token.value = savedToken
    }

    // 恢复用户信息
    const savedUserInfo = localStorage.getItem('userInfo')
    if (savedUserInfo) {
      try {
        userInfo.value = JSON.parse(savedUserInfo)
      } catch (error) {
        console.error('解析用户信息失败:', error)
      }
    }
  }

  /**
   * 刷新用户信息（从服务器获取最新信息）
   */
  async function refreshUserInfo() {
    try {
      const res = await getAuthProfile()
      const data = res as any
      if (data.success && data.data) {
        setUserInfo(data.data)
        return true
      }
      return false
    } catch (error) {
      console.error('刷新用户信息失败:', error)
      return false
    }
  }

  /**
   * 退出登录
   */
  function logout() {
    userInfo.value = null
    token.value = null
    localStorage.removeItem('userInfo')
    localStorage.removeItem('token')
    sessionStorage.removeItem('token')
  }

  return {
    userInfo,
    token,
    isLogin,
    username,
    nickname,
    setUserInfo,
    setToken,
    initUserInfo,
    refreshUserInfo,
    logout
  }
})
```

### HTTP请求拦截

**工具**: `utils/request.ts:1-103`

核心特性:
- 自动携带token
- 多语言Header (`Accept-Language`)
- 客户端标识 (`X-Client-Type: h5`)
- 401自动跳转登录

```typescript
// 请求拦截器
request.interceptors.request.use(
  (config) => {
    // 自动携带token
    const token = localStorage.getItem("token")
    if (token) {
      config.headers.Authorization = `Bearer ${token}`
    }

    // 标识客户端类型为 H5
    config.headers["X-Client-Type"] = "h5"

    // 自动携带语言参数
    const locale = localStorage.getItem("locale") || "zh-CN"
    config.headers["Accept-Language"] = locale

    return config
  }
)

// 响应拦截器
request.interceptors.response.use(
  (response) => response.data,
  (error) => {
    if (error.response) {
      const { status, data } = error.response

      switch (status) {
        case 401:
          const message = data?.message || "未授权,请重新登录"
          showToast(message)

          // 清除本地存储
          localStorage.clear()
          sessionStorage.clear()

          // 延迟跳转到登录页
          setTimeout(() => {
            router.replace("/login")
          }, 1000)
          break

        case 403:
          showToast("拒绝访问")
          break

        case 404:
          showToast("请求的资源不存在")
          break

        case 500:
          showToast("服务器错误")
          break

        default:
          showToast(data?.message || "请求失败")
      }
    }
    return Promise.reject(error)
  }
)
```

```typescript
export function useTodayData() {
  const todayData = ref<TodayData>({
    weight: null,
    exercise_duration: null,
    sleep_hours: null,
    mood: null
  })

  const hasCheckedInToday = ref(false)

  async function refreshData() {
    try {
      const { getHealthRecords } = await import('@/api/health')

      const today = new Date().toISOString().split('T')[0]
      const [healthRes, dietRes] = await Promise.all([
        getHealthRecords({ start_date: today, end_date: today, limit: 1 }),
        getDietSummary({ date: today })
      ])

      // 处理健康记录
      const healthData = (healthRes as any).data?.records || []
      if (healthData.length > 0) {
        const record = healthData[0]
        todayData.value = {
          weight: record.weight,
          exercise_duration: record.exercise_duration,
          sleep_hours: record.sleep_hours,
          mood: record.mood
        }
        hasCheckedInToday.value = true
      } else {
        hasCheckedInToday.value = false
      }
    } catch (error) {
      console.error('加载今日数据失败:', error)
    }
  }

  onMounted(() => {
    refreshData()
  })

  return { todayData, hasCheckedInToday, refreshData }
}
```
