## 1. 项目概述

移动端H5应用采用Vue 3 Composition API架构，提供健康数据记录、饮食管理、目标追踪、数据可视化等功能。

**核心特性**：
- 📱 移动优先设计 (375px基准)
- 🌍 国际化支持 (中文/English)
- 📊 ECharts数据可视化
- 🎯 Vue3 Tour新手引导
- 🔄 下拉刷新/上拉加载
- 💾 离线数据缓存

### 3.2 健康打卡模块

**核心功能**：日历视图显示打卡记录、点击日期录入数据、已打卡日期标记

#### 📅 重难点3：自定义日历网格生成算法

**挑战**：生成42格日历(6周×7天)，处理跨月日期、未来日期禁用、打卡状态标记

**解决方案**：

```typescript
// views/health/components/HealthCalendar.vue:120-183
const calendarDays = computed(() => {
  const year = currentYear.value
  const month = currentMonth.value
  const firstDay = new Date(year, month, 1)
  const firstDayOfWeek = firstDay.getDay() // 本月1号是周几
  const daysInMonth = new Date(year, month + 1, 0).getDate() // 本月天数
  const prevMonthLastDay = new Date(year, month, 0).getDate() // 上月最后一天

  const days: CalendarDay[] = []
  const today = new Date()
  today.setHours(0, 0, 0, 0)

  // 1️⃣ 填充上月尾部日期
  for (let i = firstDayOfWeek - 1; i >= 0; i--) {
    const day = prevMonthLastDay - i
    const date = new Date(year, month - 1, day)
    days.push({
      day,
      isCurrentMonth: false,
      isChecked: checkedDates.value.has(formatDate(date)),
      isFuture: date > today
    })
  }

  // 2️⃣ 填充本月日期
  for (let day = 1; day <= daysInMonth; day++) {
    const date = new Date(year, month, day)
    days.push({
      day,
      isCurrentMonth: true,
      isChecked: checkedDates.value.has(formatDate(date)),
      isFuture: date > today
    })
  }

  // 3️⃣ 填充下月开头日期，补齐到42格 (6周 × 7天)
  const remainingDays = 42 - days.length
  for (let day = 1; day <= remainingDays; day++) {
    const date = new Date(year, month + 1, day)
    days.push({
      day,
      isCurrentMonth: false,
      isChecked: checkedDates.value.has(formatDate(date)),
      isFuture: date > today
    })
  }

  return days
})
```

#### 📊 重难点5：Swiper智能居中算法

**挑战**：横向滚动的食物分类，点击后自动居中显示

**解决方案**：

```typescript
// views/diet/components/FoodCategorySwiper.vue:74-103
function handleSelectCategory(category: string, index: number) {
  emit("select", category)

  nextTick(() => {
    const swiper = categorySwiperRef.value?.$el?.swiper
    if (!swiper) return

    const clickedSlide = swiper.slides[index]
    const swiperWidth = swiper.width
    const slideLeft = clickedSlide.offsetLeft
    const slideWidth = clickedSlide.offsetWidth

    // 计算居中所需的偏移量
    const centerOffset = slideLeft - (swiperWidth / 2 - slideWidth / 2)
    const maxTranslate = swiper.virtualSize - swiperWidth

    // 边界处理
    let targetTranslate
    if (centerOffset <= 0) {
      targetTranslate = 0 // 左边界
    } else if (centerOffset >= maxTranslate) {
      targetTranslate = maxTranslate // 右边界
    } else {
      targetTranslate = centerOffset // 居中
    }

    swiper.setTranslate(-targetTranslate)
    swiper.transitionStart()
  })
}
```

```typescript
// views/history/utils/hooks.ts:11-77
export function useHealthRecords() {
  const records = ref<API.HealthRecord[]>([])
  const loading = ref(false)
  const finished = ref(false)
  const refreshing = ref(false)
  const page = ref(1)
  const pageSize = 15

  // 默认当前周
  const weekRange = getCurrentWeekRange()
  const startDate = ref<string>(weekRange.start)
  const endDate = ref<string>(weekRange.end)

  async function loadRecords() {
    if (finished.value) return
    loading.value = true

    try {
      const response = await getHealthRecords({
        page: page.value,
        limit: pageSize,
        start_date: startDate.value,
        end_date: endDate.value
      })

      const newRecords = response.data.records

      // 下拉刷新：替换；上拉加载：追加
      if (refreshing.value) {
        records.value = newRecords
        refreshing.value = false
      } else {
        records.value.push(...newRecords)
      }

      page.value++
      if (newRecords.length < pageSize) {
        finished.value = true
      }
    } catch (error) {
      console.error('加载记录失败:', error)
    } finally {
      loading.value = false
    }
  }

  function setDateRange(start: string, end: string) {
    startDate.value = start
    endDate.value = end
    page.value = 1
    finished.value = false
    records.value = []
    loadRecords()
  }

  return { records, loading, finished, refreshing, loadRecords, setDateRange }
}
```

### 7.3 核心源码 - 头像上传功能

#### 📸 功能亮点

这是移动端的核心交互功能之一，展示了完整的文件上传流程：

#### 🎯 完整实现代码

```vue
<template>
  <!-- 头像上传组件 -->
  <div class="avatar-upload-wrapper">
    <van-uploader
      v-model="fileList"
      :max-count="1"
      :max-size="5 * 1024 * 1024"
      :before-read="beforeRead"
      :after-read="afterRead"
      :deletable="false"
      preview-size="80"
      @oversize="onOversize"
    >
      <template #default>
        <div class="avatar-wrapper">
          <!-- 头像显示 -->
          <van-image
            v-if="avatarUrl && !isUploading"
            round
            :src="avatarUrl"
            fit="cover"
          >
            <template #error>
              <div class="avatar-placeholder">
                <van-icon name="user-o" size="40" />
              </div>
            </template>
          </van-image>

          <!-- 上传状态 -->
          <div v-if="isUploading" class="avatar-uploading">
            <van-loading type="spinner" size="30" color="#1989fa" />
            <div class="upload-text">{{ t("上传中") }}</div>
          </div>

          <!-- 相机图标遮罩 -->
          <div class="avatar-mask">
            <van-icon name="photograph" size="20" color="#fff" />
          </div>
        </div>
      </template>
    </van-uploader>
  </div>
</template>

<script setup lang="ts">
import { ref, onActivated } from "vue";
import { showToast } from "vant";
import type { UploaderFileListItem } from "vant";
import { useUserStore } from "@/stores/user";
import { postAuthAvatar } from "@/api/auth";
import { useI18n } from "vue-i18n";

const { t } = useI18n();
const userStore = useUserStore();
const fileList = ref<UploaderFileListItem[]>([]);
const isUploading = ref(false);
const avatarUrl = ref<string | null>(null);

// 初始化头像URL（处理开发/生产环境）
const initAvatarUrl = () => {
  const avatar = userStore.userInfo?.avatar;
  if (!avatar) return "";

  // 完整URL直接返回
  if (avatar.startsWith("http://") || avatar.startsWith("https://")) {
    return avatar;
  }

  // 生产环境拼接域名
  const staticBaseUrl = import.meta.env.VITE_STATIC_BASE_URL;
  if (staticBaseUrl) {
    return `${staticBaseUrl}${avatar}`;
  }

  // 开发环境返回相对路径（Vite代理处理）
  return avatar;
};

// 页面激活时刷新头像
onActivated(async () => {
  avatarUrl.value = initAvatarUrl();
  await userStore.refreshUserInfo();
  avatarUrl.value = initAvatarUrl();
});

// 上传前验证文件类型
function beforeRead(file: File | File[]) {
  const fileToCheck = Array.isArray(file) ? file[0] : file;
  if (!fileToCheck) return false;

  const allowedTypes = [
    "image/jpeg",
    "image/jpg",
    "image/png",
    "image/gif",
    "image/webp"
  ];

  if (!allowedTypes.includes(fileToCheck.type)) {
    showToast(t("只支持上传 JPG/PNG/GIF/WEBP 格式的图片"));
    return false;
  }
  return true;
}

// 文件超过大小限制
function onOversize() {
  showToast(t("图片大小不能超过 5MB"));
}

// 上传文件到服务器
async function afterRead(file: UploaderFileListItem | UploaderFileListItem[]) {
  const fileItem = Array.isArray(file) ? file[0] : file;
  if (!fileItem?.file) return;

  // 设置上传状态
  isUploading.value = true;
  fileItem.status = "uploading";
  fileItem.message = t("上传中...");

  try {
    // 调用上传API
    const res = await postAuthAvatar({}, fileItem.file);
    const data = res as any;

    if (data.success && data.data?.avatarUrl) {
      fileItem.status = "done";
      fileItem.message = "";

      // 更新本地store
      userStore.updateAvatar(data.data.avatarUrl);
      avatarUrl.value = initAvatarUrl();

      // 刷新完整用户信息
      await userStore.refreshUserInfo();

      showToast(t("头像上传成功"));
    } else {
      throw new Error(data.message || t("上传失败"));
    }
  } catch (error: any) {
    console.error("头像上传失败:", error);
    fileItem.status = "failed";
    fileItem.message = t("上传失败");
    showToast(error.message || t("头像上传失败，请重试"));
  } finally {
    isUploading.value = false;
    // 清空文件列表
    setTimeout(() => {
      fileList.value = [];
    }, 1000);
  }
}
</script>
```
